\begin{frame}{Odkrivanje tabele po vrsticah}
    Včasih pride prav, da tabelo odkrivamo postopoma po vrsticah.

    \begin{center}
        \begin{tabular}{c|cccc} %chktex 44
            Oznaka & A & B & C & D \\ \hline %chktex 44
            \pause{}
            X & 1 & 2 & 3 & 4 \\
            \pause{}
            Y & 3 & 4 & 5 & 6 \\
            \pause{}
            Z & 5 & 6 & 7 & 8
        \end{tabular}
    \end{center}
\end{frame}
